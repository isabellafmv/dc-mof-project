# feature-generation step (mofdscribe / pymatgen) pins an older Python
# than the modelling/analysis work. Create both:
#
#   # 1) core analysis + modelling (most notebooks)
#   conda create -n mof_ml python=3.11
#   conda activate mof_ml
#   pip install -r requirements.txt            # installs Section A + C
#
#   # 2) RACS feature generation (notebook 02 only)
#   conda create -n mofdscribe python=3.10
#   conda activate mofdscribe
#   pip install -r requirements.txt            # install Section B here
#
# Version pins are intentionally loose; pin exact versions from your
# working environments (`pip freeze`) before final submission.
